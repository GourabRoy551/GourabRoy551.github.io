\cventryshort{Cross-Modal Attention Bridge (CMAB) for CLIP Interpretability}%
             {UAM Madrid, Feb 2026 -- Ongoing}

  \item Developing a statistically filtered cross-modal attention method explaining
        CLIP's image--text alignment; K-sigma filtering on attention rollouts generates

  \item Supervised by Prof.\ Jenny Benois-Pineau (UBx Bordeaux) and
        Prof.\ Marcos Escudero-Vi\~nolo (UAM Madrid) --- \textit{TRDP~1 research project}
  \item \textit{Tools:} PyTorch $\cdot$ ViT $\cdot$ CLIP $\cdot$ Attention Rollout $\cdot$ XAI
\end{itemize}

\vspace{0.25em}

\cventryshort{Cardiac Autonomic Assessment via PPG-Derived HRV During Sound Stimulation}%
             {PPKE Budapest, Dec 2025}
\begin{itemize}
  \item Modular Python pipeline: 10 participants across baseline and music conditions;
        Butterworth filtering, adaptive peak detection, FFT spectral analysis,
        Friedman/Wilcoxon tests and individual responsiveness classifier ---
        every subject showed significant ANS shifts despite non-significant group means
  \item \href{https://github.com/GourabRoy551/cardiac-autonomic-ppg-hrv}{github.com/GourabRoy551/cardiac-autonomic-ppg-hrv}
  \item \textit{Tools:} Python $\cdot$ NumPy $\cdot$ SciPy $\cdot$ Pandas $\cdot$ Matplotlib
\end{itemize}

\vspace{0.25em}

\cventryshort{Calibrated Multi-View 3D Reconstruction}{UAM Madrid, 2025}
\begin{itemize}

        RANSAC, fundamental matrix, epipolar geometry, multi-view sparse reconstruction,

\end{itemize}
